# Tarjan

## 找割点

```cpp
void tarjan(int u,int fa){
    dfn[u]=low[u]=++tot; 
    int child=0;
    for(int i=0;i<e[u].size();i++){
        int v=e[u][i];
        if(!dfn[v]){
            tarjan(v,u);
            low[u]=min(low[u],low[v]);
            if(low[v]>=dfn[u]&&fa!=0){
                cut[u]=1;
            }
            child++;
        } 
        else if(v!=fa){
            low[u]=min(low[u],dfn[v]);
        }
    }
    if(fa==0&&child>=2){
        cut[u]=1;
    }
}
```
## 找割边

```cpp
vector<pair<int,int>> ans;
void tarjan(int u,int fa){
    dfn[u]=low[u]=++tot; 
    for(int i=0;i<e[u].size();i++){
        int v=e[u][i];
        if(!dfn[v]){
            tarjan(v,u);
            low[u]=min(low[u],low[v]);
            if(low[v]>dfn[u]){
                if(u>v) ans.push_back({v,u});
                else ans.push_back({u,v});
            }
        } 
        else if(v!=fa){
            low[u]=min(low[u],dfn[v]);
        }
    }
}
```

## 求解强连通分量模版

```cpp
void tarjan(int u){
	dfn[u]=low[u]=++tot;
	stk.push(u);
	instk[u]=1;
	for(int i=0;i<e[u].size();i++){
		int v=e[u][i];
		if(!dfn[v]){
			tarjan(v);
			low[u]=min(low[u],low[v]);
		}
		else if(instk[v]){
			low[u]=min(low[u],dfn[v]);
		}
	}
	if(dfn[u]==low[u]) {
		scccnt++;
		int v;
		do{
			v=stk.top();stk.pop();
			instk[v]=0;
			scc[v]=scccnt;
			sccsize[scccnt]++;
		} while(v!=u);
	}
}
```
