# 最近公共祖先
## 求深度并初始化f[u][0]
```cpp
void dfs(int u){
    vis[u]=1;
    for(int i=0;i<e[u].size();i++){
        if(!vis[e[u][i]]){
            dep[e[u][i]]=dep[u]+1;
            f[e[u][i]][0]=u;
            dfs(e[u][i]);    
        }    
    }
}

```

## 求LCA
```cpp
int LCA(int u,int v){
    if(dep[u]<dep[v])
        swap(u,v);
    for(int i=19;i>=0;i--)
        if(dep[u]-dep[v]>=(1<<i))
            u=f[u][i];
    if(u==v) return u;
    for(int i=19;i>=0;i--)
        if(f[u][i]!=f[v][i])
            u=f[u][i],v=f[v][i];
    return f[u][0];
}
```
