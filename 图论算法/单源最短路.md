### 链式前向星
```cpp
struct edge{
    int v,w,next;
}e[M];//M为边的数量，无向图是M*2
int cnt,head[N];//N为结点数量
void add(int u, int v,int w){
    e[++cnt].v=v;
    e[cnt].w=w;
    e[cnt].next=head[u];
    head[u]=cnt;
}
int main(){

        int u,v,w;
        cin>>u>>v>>w;

    }
```

## Bellman-Ford算法
可以处理负权边，并能检测负权环
原理对所有边进行 V-1 次松弛。
O(nm)
### 基本模版
```cpp
void Bellman_Ford(){
        for(int i=1;i<n;i++){//初始化
            d[i]=inf;
        }
        d[s]=0;
        for(int i=1;i<n;i++){
            bool flag=0;//记录本轮是否有更新
            for(int j=1;j<=m;j++){//松弛操作
                if(d[e[j].u] < inf && d[e[j].u]+e[j].w<d[e[j].v]){
                    d[e[j].v]=d[e[j].u]+e[j].w;
                    flag=1;
                }    
            }
            if(!flag) break;
        }
}

```

## SPFA算法
只处理变化的边
### 基本模版 
```cpp
void spfa(){
    for(int i=1;i<=n;i++){
        d[i]=inf;
    }
    d[s]=0;
    queue<int> q;
    q.push(s);
    vis[s]=1;
    while(!q.empty()){
        int u=q.front();
        q.pop();vis[u]=0;
        for(int i=head[u];i;i=e[i].next){
            int v=e[i].v;
            if(d[v]>d[u]+e[i].w){
                d[v]=d[u]+e[i].w;
                if(!vis[v]){
                    q.push(v);vis[v]=1;
                }
            }
        }
    }
}
```
### 带负环反馈的模版
```cpp
bool spfa(){
    for(int i=1;i<=n;i++){
        d[i]=inf;
    }
    d[s]=0;
    queue<int> q;
    q.push(s);
    vis[s]=1; in[s]++;
    while(!q.empty()){
        int u=q.front();
        q.pop();vis[u]=0;
        for(int i=head[u];i;i=e[i].next){
            int v=e[i].v;
            if(d[v]>d[u]+e[i].w){
                d[v]=d[u]+e[i].w;
                if(!vis[v]){
                    q.push(v);vis[v]=1;
                    in[v]++;
                    if(in[v]>n) return 0;
                }
            }
        }
    }
    return 1;
}
```

## Dijkstra算法
维护一个集合 S：已确定最短路径的节点

每次从未处理的节点中，选择距离源点最近的节点 u

将 u 加入 S
O（(n+m)logn）
用 u 松弛它的所有出边（更新邻居的距离）
### 基本模版
```cpp
for(int i = 1;i <= n;i++){
        for(int j = 1;j <= n;j++){
            if(i==j){
                e[i][j] = 0;
            }else{
                e[i][j] = inf;
            }
        }
    }
    vis[s] = 1;
    for(int i=1;i<=m;i++){
        cin >> p >> q >> t;
        e[p][q] = t; 
    }
for(int i = 1;i <= n;i++){    
        d[i] = e[s][i]; 
    }
    for(int i = 1;i <= n-1;i++){
        min = inf;
        for(int j = 1;j <= n;j++){
            if(vis[j] == 0&&d[j] < min){
                min = d[j];
                u = j;
            }
        }
        vis[u] = 1;
        for(int v=1;v<=n;v++){
            if(e[u][v] < inf){
                if(d[v] > d[u] + e[u][v]){
                    d[v] = d[u] + e[u][v]; 
                }
            }
        }
} 
```
### 优化
```cpp
void dijkstra(){
    for(int i=1;i<=n;i++){//初始化
        d[i]=inf;
    }
    d[s] = 0;
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;//优先队列，存储pair<距离, 节点编号>，使用greater实现小根堆
    pq.push({0, s});
    while (!pq.empty()) {
        int u = pq.top().second;  // 当前节点
        pq.pop();
        if (vis[u]) continue;
        vis[u] = 1;
        for (int i = head[u]; i; i = e[i].next) {
            int v = e[i].v;
            int w = e[i].w;
            if (!vis[v] && d[v] > d[u] + w) {//松弛操作
                d[v] = d[u] + w;
                pq.push({d[v], v});
            }
        }
    }
}
```
