## 模版二：使用动态数组存储图，用队列的STL模版

```cpp
vector<int> e[105];
queue<int> q;
for(int i=1;i<=n;i++){
    if(in[i]==0){
        q.push(i);            
    }
}
while(!q.empty()){
    int u=q.front();
    cout<<u<<" ";
    q.pop();
    for(int i=0;i<e[u].size();i++){
        int v=e[u][i];
        in[v]--;
        if(in[v]==0)
            q.push(v);
    }
}
```
