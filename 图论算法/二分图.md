# 二分图判定基本模版
[^1]: 可以用黑白染色，可以分成两部分，部分内没有边
```cpp
struct edge{
    int v,next;
}e[M];//M为边的数量，无向图是M*2
int cnt,head[N];//N为结点数量
void add(int u, int v){
    e[++cnt].v=v;
    e[cnt].next=head[u];
    head[u]=cnt;
}
int c[N];//N为结点数量
bool dfs(int u){
    for(int i=head[u];i;i=e[i].next){
        int v=e[i].v;
        if(c[v]==-1){
            c[v]=!c[u];
            if(!dfs(v))
                return 0;
        }
        else if(c[u]==c[v])
            return 0;
    }
    return 1;
}
```
