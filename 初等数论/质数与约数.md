## 欧筛
```cpp
void prime(n){
    for (int i=2; i<=n; i++) {
        if (a[i]==0) {
            b[num++]=i; //保存质数
        }
        for (int j=0; j<num; j++) { //遍历现有质数
            if (i*b[j]>n) break; //合数超过最大范围
            a[i*b[j]]=1;//标记合数
            if (i%b[j]==0) break; //后续质数不是合数的最小质因子
        }
    }
}
```
# 题目模版
## 阶乘的质数分解
### 分解质因数方法
```cpp
#include <bits/stdc++.h> 
using namespace std;
int cnt[10005];
void factor(int n){
	for(int i=2;i*i<=n;i++){
		while(n%i==0){
			n/=i;
			cnt[i]++;
		}    
	}
	if(n!=1){
		cnt[n]++;
	}
}
int main(){
	int n;
	cin>>n;
	for(int i=2;i<=n;i++){
		factor(i);	
	}
	for(int i=2;i<=n;i++){
		if(cnt[i]) cout<<i<<" "<<cnt[i]<<endl;	
	}
	return 0;	
}
```
### 使用欧筛方法
```cpp
#include<bits/stdc++.h>
using namespace std;
int a[10005],b[10005],n,num;
void prime() {
	for (int i=2; i<=n; i++) {
		if (a[i]==0) {
			b[num++]=i;
		}
		for (int j=0; j<num; j++) {
			if (i*b[j]>n) break;
			a[i*b[j]]=1;
			if (i%b[j]==0) break;
		}
	}
}
int main(){
	cin>>n;
	prime();
	for(int i=0;i<num;i++){
		int cnt = 0;
		int temp=n;
		while(temp){
			cnt+=temp/b[i];
			temp/=b[i];
		}
		cout<<b[i]<<" "<<cnt<<endl; 
	} 
	return 0;	
}
```
## 最大约数和
```cpp
#include <bits/stdc++.h> 
using namespace std;
int sumdiv[1005];
int dp[1005];
int main() {
    int s;
    cin >> s;
    // 筛法计算真约数和，时间复杂度O(slogs)
    for (int i = 1; i <= s/2; i++) {
        for (int j = 2 * i; j <= s; j += i) {
            sumdiv[j] += i;
        }
    }    
    // 0/1 背包
    for (int i = 1; i <= s; i++) {
        for (int j = s; j >= i; j--) {
            dp[j] = max(dp[j], dp[j - i] + sumdiv[i]);
        }
    }
    cout << dp[s] << endl;
    return 0;
}
```
