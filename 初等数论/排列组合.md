# 求组合数
## 递推O(n<sup>2</sup>)
```cpp
c[0][0]=1;
    for(int i=1;i<=n;i++){
        c[i][0]=1;
        for(int j=1;j<=m;j++){
            c[i][j]=(c[i-1][j-1]+c[i-1][j]) % MOD;
        }
    }    
```
## 直接求阶乘逆元O(n+log p)

```cpp
fact[0] = 1;
    for (int i = 0; i < n; i++) {//阶乘取余 
        fact[i + 1] = fact[i] * (i + 1) % MOD;
    }
    inv[n] = qpow(fact[n], MOD - 2);  // 费马小定理
    inv[m] = qpow(fact[m], MOD - 2);  // 费马小定理
    inv[n-m] = qpow(fact[n-m], MOD - 2);  // 费马小定理
    ll c = fact[n] * inv[m] % MOD * inv[n - m] % MOD;
```

## 递推求阶乘逆元

```cpp
fact[0] = 1;
    for (int i = 0; i < n; i++) {//阶乘取余 
        fact[i + 1] = fact[i] * (i + 1) % MOD;
    }
    inv[n] = qpow(fact[n], MOD - 2);  // 费马小定理
    for (int i = n - 1; i >= 0; i--) {
        inv[i] = inv[i + 1] * (i + 1) % MOD;
    }   
    ll C(ll n, ll m) { // 计算组合数
    if (n < m)
        return 0;
    if (m == 0)
        return 1;
    return fact[n] * inv[m] % MOD * inv[n - m] % MOD;
    }

```
## 递推求阶乘逆元 O(m+log p)
```cpp
fact[0] = 1;
    for (int i = 0; i < m; i++) {//阶乘取余 
        fact[i + 1] = fact[i] * (i + 1) % MOD;
    }
    inv[m] = qpow(fact[m], MOD - 2);  // 费马小定理
    for (int i = m - 1; i >= 0; i--) {
        inv[i] = inv[i + 1] * (i + 1) % MOD;
    }    
    ll C(ll n, ll m) { // 计算组合数
    if (n < m)
        return 0;
    if (m == 0)
        return 1;
    ll num = 1;
    for (int i = 0; i < m; i++) num = num * ((n - i) % MOD)% MOD;
    return num * inv[m] % MOD;
    }

```
