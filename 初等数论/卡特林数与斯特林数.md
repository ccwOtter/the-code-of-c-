c[1]=1;

## 通项+逆元

```cpp
l fact[40], inv[40];
ll qpow(ll a, ll n){
    if(n == 0)    return 1;
    ll x = qpow(a, n/2);
    ll ans = x * x % MOD;
    if(n % 2 == 1)    ans = ans * a % MOD;
    return ans;
}
int main() {
    int n;
    cin >> n;
    fact[0] = 1;
    for (int i = 0; i < 2 * n; i++) {//阶乘取余 
        fact[i + 1] = fact[i] * (i + 1) % MOD;
    }
    inv[n] = qpow(fact[n], MOD - 2);  // 费马小定理
    inv[n+1] = qpow(fact[n+1], MOD - 2);  // 费马小定理
    cout << fact[2*n] * inv[n] % MOD * inv[n+1] % MOD;
```
