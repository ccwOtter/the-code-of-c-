
# 线段树模板

## 1、结构体表示结点
```cpp
struct treenode {
    int l, r;
    long long s, tag;
} tree[4 * maxn];
```
## 2、构建线段树

```cpp
void build(int i, int l, int r) {
    tree[i].l = l; tree[i].r = r; tree[i].tag = 0;
    if (l == r) { tree[i].s = a[l]; return; }
    int mid = (l + r) / 2;
    build(i * 2, l, mid);
    build(i * 2 + 1, mid + 1, r);
    tree[i].s = tree[i * 2].s + tree[i * 2 + 1].s;
}
```
## 懒惰标记

```cpp
void push(int i) {
    if (tree[i].tag) {
        tree[i * 2].s += tree[i].tag * (tree[i * 2].r - tree[i * 2].l + 1);
        tree[i * 2 + 1].s += tree[i].tag * (tree[i * 2 + 1].r - tree[i * 2 + 1].l + 1);
        tree[i * 2].tag += tree[i].tag;
        tree[i * 2 + 1].tag += tree[i].tag;
        tree[i].tag = 0;
    }
}
```
## 4、区间修改
```cpp
void update(int i, int l, int r, long long val) {
    if (l <= tree[i].l && tree[i].r <= r) {
        tree[i].s += val * (tree[i].r - tree[i].l + 1);
        tree[i].tag += val;
        return;
    }
    push(i);
    int mid = (tree[i].l + tree[i].r) / 2;
    if (l <= mid) update(i * 2, l, r, val);
    if (r > mid) update(i * 2 + 1, l, r, val);
    tree[i].s = tree[i * 2].s + tree[i * 2 + 1].s;
}
```
## 5、区间查询
```cpp
long long ask(int i, int l, int r) {
    if (l <= tree[i].l && tree[i].r <= r) return tree[i].s;
    push(i);
    int mid = (tree[i].l + tree[i].r) / 2;
    long long ans = 0;
    if (l <= mid) ans += ask(i * 2, l, r);
    if (r > mid) ans += ask(i * 2 + 1, l, r);
    return ans;
}
```
